"Show me the top 5 users by spend in the last 30 days, grouped by region."
1 SELECT u.name, SUM(t.amount) AS total_spend
2 FROM users u
3 JOIN transactions t ON u.id = t.user_id
4 WHERE t.created_at >= NOW() - INTERVAL '30 days'
5 GROUP BY u.region, u.name
6 ORDER BY total_spend DESC
7 LIMIT 5;
