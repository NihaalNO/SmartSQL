query: "Show me monthly revenue by region for 2023"
SELECT
DATE_TRUNC('month', order_date) AS month,
region,
SUM(revenue) AS total_revenue
FROM sales_warehouse.orders
WHERE EXTRACT(year FROM order_date) = 2023
GROUP BY 1, 2
ORDER BY 1, 2;
